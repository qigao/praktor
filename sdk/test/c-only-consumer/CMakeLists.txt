cmake_minimum_required(VERSION 3.20)
project(PraktorC11ExternalConsumer LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

add_test(NAME praktor_c11_external_consumer COMMAND praktor_c11_external_consumer)
