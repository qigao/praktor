cmake_minimum_required(VERSION 3.27)
project(PraktorC11SessionStandalone LANGUAGES C)
include(CTest)
find_package(SaltsUtils CONFIG REQUIRED
    PATHS "$ENV{SALTS_UTILS_ROOT}/lib/cmake/SaltsUtils" NO_DEFAULT_PATH)
foreach(required IN ITEMS Salts::BehaviorTree Salts::Process Salts::JsonParser Salts::TinyTest)
    if(NOT TARGET ${required})
        message(FATAL_ERROR "C11 Session requires ${required}")
    endif()
endforeach()
add_library(praktor_c11_bt_core STATIC ../tree_adapter.c)
target_include_directories(praktor_c11_bt_core PUBLIC ..)
target_link_libraries(praktor_c11_bt_core PUBLIC Salts::BehaviorTree)
add_library(praktor_c11_value_store STATIC ../value_store.c)
target_include_directories(praktor_c11_value_store PUBLIC ..)
foreach(target IN ITEMS praktor_c11_bt_core praktor_c11_value_store)
    target_compile_features(${target} PUBLIC c_std_11)
    set_target_properties(${target} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
endforeach()
include(../session_core_targets.cmake)
if(CMAKE_CXX_COMPILER_LOADED OR TARGET actions OR TARGET praktor_actions_c11 OR TARGET praktor_c11_session OR TARGET praktor_shell)
    message(FATAL_ERROR "C11 Session must not load a C++ compiler or execution target")
endif()
add_executable(actions_c11_session_c_test ../session_c_test.c)
target_link_libraries(actions_c11_session_c_test PRIVATE praktor_c11_session_core praktor_c11_native_host Salts::TinyTest)
set_target_properties(actions_c11_session_c_test PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF LINKER_LANGUAGE C)
add_test(NAME actions_c11_session_c_test COMMAND actions_c11_session_c_test)
set_tests_properties(actions_c11_session_c_test PROPERTIES TIMEOUT 30)

add_executable(actions_c11_tree_plan_test ../plan_test.c)
target_link_libraries(actions_c11_tree_plan_test PRIVATE praktor_c11_session_core Salts::TinyTest)
set_target_properties(actions_c11_tree_plan_test PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF LINKER_LANGUAGE C)
add_test(NAME actions_c11_tree_plan_test COMMAND actions_c11_tree_plan_test)
