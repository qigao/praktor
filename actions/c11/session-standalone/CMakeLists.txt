cmake_minimum_required(VERSION 3.27)
project(PraktorC11IndependentSession LANGUAGES C CXX)
include(CTest)
find_package(SaltsUtils CONFIG REQUIRED
    PATHS "$ENV{SALTS_UTILS_ROOT}/lib/cmake/SaltsUtils" NO_DEFAULT_PATH)
foreach(required IN ITEMS Salts::BehaviorTree Salts::Process Salts::JsonParser)
    if(NOT TARGET ${required})
        message(FATAL_ERROR "Independent Session requires ${required}")
    endif()
endforeach()
if(TARGET actions OR TARGET praktor_actions_c11 OR TARGET praktor_shell)
    message(FATAL_ERROR "Legacy execution targets are forbidden in this project")
endif()
add_library(praktor_c11_bt_core STATIC ../tree_adapter.c)
target_include_directories(praktor_c11_bt_core PUBLIC ..)
target_link_libraries(praktor_c11_bt_core PUBLIC Salts::BehaviorTree)
add_library(praktor_c11_value_store STATIC ../value_store.c)
target_include_directories(praktor_c11_value_store PUBLIC ..)
foreach(target IN ITEMS praktor_c11_bt_core praktor_c11_value_store)
    target_compile_features(${target} PUBLIC c_std_11)
    set_target_properties(${target} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
endforeach()
include(../session_targets.cmake)
add_executable(actions_c11_session_independent_test ../session_independent_test.cpp)
target_link_libraries(actions_c11_session_independent_test PRIVATE praktor_c11_session praktor_c11_native_host)
add_test(NAME actions_c11_session_independent_test COMMAND actions_c11_session_independent_test)
set_tests_properties(actions_c11_session_independent_test PROPERTIES TIMEOUT 30)
