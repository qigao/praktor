cmake_minimum_required(VERSION 3.27)
project(PraktorOutputStandalone LANGUAGES C)
include(CTest)
find_package(Salts CONFIG REQUIRED PATHS "$ENV{SALTS_ROOT}" NO_DEFAULT_PATH)
include(../output_targets.cmake)
if(CMAKE_CXX_COMPILER_LOADED OR TARGET actions OR TARGET Salts::JsonParser OR TARGET Salts::Process)
    message(FATAL_ERROR "Output scanner must use only the base C SDK")
endif()
add_executable(actions_output_core_test ../core/output_scan_test.c)
target_include_directories(actions_output_core_test PRIVATE ../core)
target_link_libraries(actions_output_core_test PRIVATE praktor_output_core Salts::TinyTest)
set_target_properties(actions_output_core_test PROPERTIES C_STANDARD 11
    C_STANDARD_REQUIRED ON C_EXTENSIONS OFF LINKER_LANGUAGE C)
add_test(NAME actions_output_core_test COMMAND actions_output_core_test)
