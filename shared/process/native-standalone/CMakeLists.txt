cmake_minimum_required(VERSION 3.27)
project(PraktorNativeApplicationStandalone LANGUAGES C)
include(CTest)
find_package(Salts CONFIG REQUIRED PATHS "$ENV{SALTS_ROOT}" NO_DEFAULT_PATH)
include(../native_target.cmake)
include(../service_targets.cmake)
include(../service_tests.cmake)
if(CMAKE_CXX_COMPILER_LOADED OR TARGET actions OR TARGET praktor_shell OR TARGET Salts::Process)
    message(FATAL_ERROR "Native application C project must not load a C++ or Salts::Process backend")
endif()
foreach(kind control native)
    add_executable(praktor_process_${kind}_test ../test/process_${kind}_test.c)
    target_link_libraries(praktor_process_${kind}_test PRIVATE praktor_process_native Salts::TinyTest Salts::Platform)
    set_target_properties(praktor_process_${kind}_test PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF LINKER_LANGUAGE C)
    add_test(NAME praktor_process_${kind}_test COMMAND praktor_process_${kind}_test)
    set_tests_properties(praktor_process_${kind}_test PROPERTIES TIMEOUT 30)
endforeach()
if(WIN32)
    add_executable(praktor_native_fixture ../../../praktor/test/helpers/managed_process_fixture.c)
    set_target_properties(praktor_native_fixture PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF LINKER_LANGUAGE C)
    add_dependencies(praktor_process_native_test praktor_native_fixture)
    target_compile_definitions(praktor_process_native_test PRIVATE PRAKTOR_NATIVE_FIXTURE="$<TARGET_FILE:praktor_native_fixture>")
endif()
